# Manual steps for the `laptop` profile: the application of its system/
# material, which phios-install only ever shows (`--system-diff`) and never
# touches. The `sudo install` lines assume the current directory is the
# phios-dotfiles checkout root; read each source first.

# --- BF-3 round 2: NetworkManager → resolvconf, for WireGuard DNS ---------
# Routes NM's DNS through resolvconf (openresolv, already installed) so
# openresolv is the single owner of /etc/resolv.conf. Prerequisite for
# round 3 below — do this one first. Read the file's own header. Reload NM
# after.
sudo install -Dm644 profiles/laptop/system/etc/NetworkManager/conf.d/rc-manager.conf /etc/NetworkManager/conf.d/rc-manager.conf
sudo systemctl reload NetworkManager.service

# --- BF-3 round 3: Tailscale → resolvconf mode ---------------------------
# Round 2 made NetworkManager feed openresolv, but the VPN toggle still
# broke. Cause: `tailscaled`'s DNS manager runs in "direct" mode — it owns
# /etc/resolv.conf outright, holds an inotify watch on it, and reverts
# every other writer (wg-quick's `resolvconf -a` never sticks). In direct
# mode it also keeps whatever nameserver it last saw there as its own
# upstream, so after `wg-quick down` it still forwards to the tunnel's now
# dead resolver → no DNS until the VPN is back up.
#
# Fix: get `tailscaled` onto resolvconf mode too, so NetworkManager,
# wg-quick and Tailscale all merge through openresolv. `tailscaled` chooses
# its DNS manager once, at start, from /etc/resolv.conf's apparent owner —
# so start it while openresolv owns the file. No file to install, and it is
# self-sustaining afterwards (openresolv writes its own header every boot).
sudo systemctl stop tailscaled
sudo resolvconf -u                          # with tailscaled stopped this now sticks
head -3 /etc/resolv.conf                     # must read "# Generated by resolvconf" before the next line
sudo systemctl start tailscaled

# Verify the mode flipped:
journalctl -u tailscaled -b --no-pager | grep -i 'dns:' | tail -5
head -3 /etc/resolv.conf                    # openresolv's header now, not "generated by tailscale"
resolvconf -l | grep -i tailscale           # non-empty: tailscaled registers through openresolv

# manual `resolvconf -u` should be needed and internet must keep working
# after down. Capture resolv.conf state both ways:
