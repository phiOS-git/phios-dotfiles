# Arch does not auto-enable user units when a package is installed, so these
# three stay off until they are enabled once, by hand.
pipewire.service
pipewire-pulse.service
wireplumber.service

# AI agent credential broker. One instance per profile class: a1 for
# general/academic/inline, a2 for coding. Enable by hand after setup, as
# described in ~/.config/phi-agent/README.md:
#   systemctl --user enable --now phi-agent-broker@a1.service
phi-agent-broker@a1.service
phi-agent-broker@a2.service

# The agent API (`phi agent serve`, contained `pi --mode rpc` per session).
# Pulls in phi-agent-broker@a1 via Wants=/After=. Serves the shell panel's
# chat (general, academic); coding never goes through it.
#   systemctl --user enable --now phi-agent.service
phi-agent.service

# Coding's egress whitelist. Both are only needed when the coding profile
# runs (`phi-code` / `phi agent code`); phi-agent-net-bridge pulls in
# phi-agent-proxy via Requires=.
#   systemctl --user enable --now phi-agent-proxy.service phi-agent-net-bridge.service
phi-agent-proxy.service
phi-agent-net-bridge.service
