# Manual steps for the `desktop` profile: the application of its system/
# material, which phios-install only ever shows (`--system-diff`) and never
# touches. The `sudo install` lines assume the current directory is the
# phios-dotfiles checkout root; read each source first.

# --- S-34: PAM service for the phi-shell lock screen ------------------------
# Lock/Lock.qml's PamContext.config is "phi-shell-lock". Without this file
# PamContext.start() fails (PamError.StartFailed) and the lock screen can
# never authenticate — it stays locked. See the file's own header.
sudo install -Dm644 profiles/desktop/system/etc/pam.d/phi-shell-lock /etc/pam.d/phi-shell-lock

# --- S-53 / OOP-33: the Plymouth boot theme --------------------------------
# The Role-A boot splash (mark + LUKS passphrase box). Install the theme
# directory, make it the default, and rebuild the initramfs so it lands in
# early boot.
sudo install -Dm644 profiles/desktop/system/usr/share/plymouth/themes/phi/phi.plymouth /usr/share/plymouth/themes/phi/phi.plymouth
sudo install -Dm644 profiles/desktop/system/usr/share/plymouth/themes/phi/phi.script   /usr/share/plymouth/themes/phi/phi.script
sudo install -Dm644 profiles/desktop/system/usr/share/plymouth/themes/phi/phi.png      /usr/share/plymouth/themes/phi/phi.png
sudo plymouth-set-default-theme -R phi
# `plymouth-set-default-theme -R` rebuilds the initramfs itself. It only
# works if `plymouth` is in the HOOKS line of /etc/mkinitcpio.conf (after
# `base udev`, before `autodetect`); add it there first if it is missing,
# then re-run the command above.
grep -q '\bplymouth\b' /etc/mkinitcpio.conf || echo 'ADD plymouth to HOOKS in /etc/mkinitcpio.conf, then re-run plymouth-set-default-theme -R phi'

# --- WireGuard control for `phi vpn` (settings-overhaul / shell-features) ---
# Lets the desktop user bring the phiOS WireGuard tunnels up and down
# without a password, so the Connectivity section's VPN toggle and
# `phi vpn up/down` work non-interactively. Without it, up/down fail with a
# clear "is 49-phi-vpn installed?" message (never a silent no-op). Read the
# file first (it documents its own scope and the accepted single-user risk),
# then install it 0440 root:root and verify it with visudo before trusting
# it. `%wheel` form assumes the desktop user is in the wheel group.
sudo install -m 0440 -o root -g root profiles/desktop/system/etc/sudoers.d/49-phi-vpn /etc/sudoers.d/49-phi-vpn
sudo visudo -cf /etc/sudoers.d/49-phi-vpn

# --- nftables firewall for `phi firewall` (Out-of-plan: firewall) ----------
# The sudoers drop-in lets the desktop user load and read the phiOS firewall
# without a password (`phi firewall` and the Connectivity > Firewall
# section). Read the file first (it documents its own tight scope and the
# accepted single-user risk). The nftables.conf below is the "firewall off"
# baseline phi manages from then on; `phi firewall enable` takes over.
sudo install -m 0440 -o root -g root profiles/desktop/system/etc/sudoers.d/49-phi-firewall /etc/sudoers.d/49-phi-firewall
sudo visudo -cf /etc/sudoers.d/49-phi-firewall
sudo install -Dm644 profiles/desktop/system/etc/nftables.conf /etc/nftables.conf
sudo systemctl enable nftables.service
# From here /etc/nftables.conf is rewritten by `phi firewall` on every
# command, so `phios-install --system-diff` will report it as differing
# whenever the firewall is enabled — that is expected, it means "the
# firewall is on", not "config drift".
