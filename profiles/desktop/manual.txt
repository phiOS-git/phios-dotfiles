# Manual steps for the `desktop` profile: the application of its system/
# material, which phios-install only ever shows (`--system-diff`) and never
# touches. The `sudo install` lines assume the current directory is the
# phios-dotfiles checkout root; read each source first.

# --- PAM service for the phi-shell lock screen ------
# PamContext.config is "phi-shell-lock". Without this, the lock screen
# cannot authenticate and stays locked.
sudo install -Dm644 profiles/desktop/system/etc/pam.d/phi-shell-lock /etc/pam.d/phi-shell-lock

# --- Plymouth boot theme -----------------------------------------------
# The boot splash (mark + LUKS passphrase box). Install the theme directory,
# make it the default, and rebuild the initramfs.
sudo install -Dm644 profiles/desktop/system/usr/share/plymouth/themes/phi/phi.plymouth /usr/share/plymouth/themes/phi/phi.plymouth
sudo install -Dm644 profiles/desktop/system/usr/share/plymouth/themes/phi/phi.script   /usr/share/plymouth/themes/phi/phi.script
sudo install -Dm644 profiles/desktop/system/usr/share/plymouth/themes/phi/phi.png      /usr/share/plymouth/themes/phi/phi.png
sudo plymouth-set-default-theme -R phi
# Ensure `plymouth` is in the HOOKS line of /etc/mkinitcpio.conf (after
# `base udev`, before `autodetect`); add it there first if missing.
grep -q '\bplymouth\b' /etc/mkinitcpio.conf || echo 'ADD plymouth to HOOKS in /etc/mkinitcpio.conf, then re-run plymouth-set-default-theme -R phi'

# --- WireGuard control for `phi vpn` -------------------------------------
# Lets the desktop user bring WireGuard tunnels up and down without a
# password. Read the sudoers file first (it documents its scope and
# single-user risk), install it 0440 root:root, and verify with visudo.
sudo install -m 0440 -o root -g root profiles/desktop/system/etc/sudoers.d/49-phi-vpn /etc/sudoers.d/49-phi-vpn
sudo visudo -cf /etc/sudoers.d/49-phi-vpn

# --- nftables firewall for `phi firewall` --------------------------------
# The sudoers drop-in lets the desktop user load and read the firewall
# without a password. Read the file first. The nftables.conf below is the
# "firewall off" baseline; `phi firewall enable` takes over.
sudo install -m 0440 -o root -g root profiles/desktop/system/etc/sudoers.d/49-phi-firewall /etc/sudoers.d/49-phi-firewall
sudo visudo -cf /etc/sudoers.d/49-phi-firewall
sudo install -Dm644 profiles/desktop/system/etc/nftables.conf /etc/nftables.conf
sudo systemctl enable nftables.service
# From here /etc/nftables.conf is rewritten by `phi firewall` on every
# command, so `phios-install --system-diff` will report it as differing
# whenever the firewall is enabled — that is expected, it means "the
# firewall is on", not "config drift".

# --- hwmon PWM fan control for `phi fan` --------------------------------
# The sudoers drop-in lets the desktop user write PWM duty cycles/enable
# modes without a password. Read the file first (it documents the hwmonN
# index instability rationale and single-user risk). Harmless to install
# on a host with no PWM fans (`phi fan status` reports plainly when none exist).
sudo install -m 0440 -o root -g root profiles/desktop/system/etc/sudoers.d/49-phi-fan /etc/sudoers.d/49-phi-fan
sudo visudo -cf /etc/sudoers.d/49-phi-fan
